Append directly below (keep one blank line between blocks):

```zsh
# --- WezTerm OSC 133 (semantic prompt zones) ----------------------------------
# FinalTerm-style prompt marks: A = prompt start, B = prompt end / input
# start, C = command output start, D;<exit> = command finished. WezTerm uses
# them for CTRL+SHIFT+Up/Down scroll-to-prompt and CTRL+triple-click
# "select this command's output" (see wezterm.lua). Harmless on terminals
# that don't speak OSC 133 — ignored, like the OSC 7 above. Inside Zellij
# the marks land in Zellij's own screen, so the wezterm-side features simply
# don't apply there (same boundary as the wezterm line-count footer).
# PARITY: mirrored in dot_bashrc.tmpl — change both together.
#
# Hook order matters twice:
#   * __wezterm_osc133_pre must run FIRST in precmd_functions — later hooks
#     (starship, mise, __wezterm_osc7) clobber $?, and D needs the real exit
#     code. add-zsh-hook only appends, so it is manually PREPENDED (with a
#     membership guard for re-sourcing), and it returns that code so
#     starship's own $? capture still sees the truth.
#   * __wezterm_osc133_post must run LAST — starship's PROMPT is a static
#     promptsubst string set at init, but appending AFTER starship's hook
#     stays correct even if a future starship rebuilds PROMPT per cycle;
#     the substring guard makes the append idempotent either way.
__wezterm_osc133_pre() {
  local last=$?
  if [[ -n ${__wezterm_cmd_ran:-} ]]; then
    printf '\033]133;D;%d\033\\' "$last"
    unset __wezterm_cmd_ran
  fi
  printf '\033]133;A\033\\'
  return $last
}
__wezterm_osc133_post() {
  # %{...%} tells ZLE the bytes are zero-width so cursor math stays right.
  if [[ ${PROMPT-} != *']133;B'* ]]; then
    PROMPT="${PROMPT}%{"$'\033]133;B\033\\'"%}"
  fi
}
__wezterm_osc133_preexec() {
  __wezterm_cmd_ran=1
  printf '\033]133;C\033\\'
}
if (( ! ${precmd_functions[(Ie)__wezterm_osc133_pre]} )); then
  precmd_functions=(__wezterm_osc133_pre "${precmd_functions[@]}")
fi
add-zsh-hook -Uz precmd __wezterm_osc133_post
add-zsh-hook -Uz preexec __wezterm_osc133_preexec
```

Append directly below (keep one blank line between blocks):

```bash
# --- WezTerm OSC 133 (semantic prompt zones) ----------------------------------
# PARITY: mirrors dot_zshrc.tmpl — change both together.
# A = prompt start, B = prompt end / input start, C = command output start,
# D;<exit> = command finished. Consumed by WezTerm for scroll-to-prompt and
# select-command-output; ignored by other terminals; inert inside Zellij.
#
# __wezterm_osc133_pre is PREPENDED to PROMPT_COMMAND: later entries
# (starship_precmd, __wezterm_osc7) clobber $?, and D needs the real exit
# code — it returns that code so starship still sees it. The B mark is
# appended to PS1 by __wezterm_osc133_post, which runs LAST: starship
# rebuilds PS1 every cycle in bash, so the append must re-apply after it
# (the substring guard keeps a static, starship-less PS1 from growing).
# C fires from PS0 (expanded once per command: after Enter, before exec).
# The '${TERM:$((__wezterm_cmd_ran=1)):0}' tail is a standard PS0 trick —
# the arithmetic ASSIGNS the flag in the current shell while the expansion
# itself produces an empty string (zero-length slice), so nothing prints.
__wezterm_osc133_pre() {
  local last=$?
  if [[ -n ${__wezterm_cmd_ran:-} ]]; then
    printf '\033]133;D;%d\033\\' "$last"
    unset __wezterm_cmd_ran
  fi
  printf '\033]133;A\033\\'
  return "$last"
}
__wezterm_osc133_post() {
  if [[ ${PS1-} != *']133;B'* ]]; then
    PS1="${PS1}\[\e]133;B\a\]"
  fi
}
case ":${PROMPT_COMMAND:-}:" in
  *:__wezterm_osc133_pre:*) ;;
  *) PROMPT_COMMAND="__wezterm_osc133_pre${PROMPT_COMMAND:+;$PROMPT_COMMAND};__wezterm_osc133_post" ;;
esac
case "${PS0:-}" in
  *']133;C'*) ;;
  *) PS0="${PS0}"'\e]133;C\a${TERM:$((__wezterm_cmd_ran=1)):0}' ;;
esac
```

  {
    event = { Drag = { streak = 1, button = 'Left' } },

  -- CTRL+triple-click — select ONE command's entire output as a unit
  -- (SemanticZone, via the OSC 133 marks from the managed rcs). The Up half
  -- completes the selection and auto-copies via copy_and_announce, mirroring
  -- the plain drag-release copy UX above. Plain triple-click keeps its
  -- default line-select. Inert where no marks exist (Zellij tabs).
  {
    event = { Down = { streak = 3, button = 'Left' } },
    mods = 'CTRL',
    action = act.SelectTextAtMouseCursor 'SemanticZone',
  },
  {
    event = { Up = { streak = 3, button = 'Left' } },
    mods = 'CTRL',
    action = act.Multiple {
      act.CompleteSelection 'PrimarySelection',
      copy_and_announce,
    },
  },
}
